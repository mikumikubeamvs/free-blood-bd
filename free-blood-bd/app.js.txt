async function loadGeoJSON(){

try{

const response =
await fetch('./bgd_admin0.geojson');

if(!response.ok){
throw new Error('GeoJSON missing');
}

const geojson =
await response.json();

geoLayer =
L.geoJSON(geojson,{

style:{
color:'#fff',
weight:1,
fillColor:'#b30000',
fillOpacity:.6
},

onEachFeature:(feature,layer)=>{

layer.on('click',()=>{

const district =
feature.properties.name ||
feature.properties.NAME_1 ||
feature.properties.admin ||
'Unknown';

document.getElementById('district').value =
district;

loadDonors(district);

});

}

}).addTo(map);

map.fitBounds(
geoLayer.getBounds()
);

}catch(err){

console.error(err);

alert(
'GeoJSON failed to load'
);

}

}

(await supabase.auth.getUser()).data.user;

}

navigator.geolocation.getCurrentPosition(

async(pos)=>{

const donor = {

document.getElementById('name').value,

document.getElementById('phone').value,

}

);

}

async function loadDonors(
districtFilter=''
){

donorMarkers.forEach(
m=>map.removeLayer(m)
);

donorMarkers=[];

const blood =
document.getElementById('searchBlood').value;

let query =
supabase
.from('donors')
.select('*')
.eq('available',true);

if(blood){

query =
query.eq(
'blood_group',
blood
);

}

if(districtFilter){

query =
query.eq(
'district',
districtFilter
);

}

const { data,error } =
await query;

const results =
document.getElementById('results');

results.innerHTML =
'<div class="loading">Loading donors...</div>';

if(error){

results.innerHTML =
'Database error';

return;

}

results.innerHTML='';

if(!data.length){

results.innerHTML =
'No donors found';

return;

}

data.forEach(d=>{

results.innerHTML += `
<div class="donor">

<b>${d.name}</b><br>

Blood:
${d.blood_group}<br>

Phone:
${d.phone}<br>

District:
${d.district}<br>

<div class="online">
${d.status}
</div>

</div>
`;

if(
d.latitude &&
d.longitude
){

const marker =
L.marker([
d.latitude,
d.longitude
])
.addTo(map)
.bindPopup(
`${d.name}
${d.blood_group}`
);

donorMarkers.push(marker);

}

});

}

async function findNearby(){

navigator.geolocation.getCurrentPosition(

async(pos)=>{

const lat =
pos.coords.latitude;

const lon =
pos.coords.longitude;

map.setView([lat,lon],11);

const { data,error } =
await supabase.rpc(
'nearby_donors',
{
lat,
lon,
radius_km:30
}
);

const results =
document.getElementById('results');

results.innerHTML='';

if(error){

results.innerHTML =
'Nearby search failed';

return;

}

if(!data.length){

results.innerHTML =
'No nearby donors';

return;

}

data.forEach(d=>{

results.innerHTML += `
<div class="donor">

<b>${d.name}</b><br>

Blood:
${d.blood_group}<br>

Phone:
${d.phone}<br>

District:
${d.district}<br>

Distance:
${Number(
d.distance_km
).toFixed(1)} km

</div>
`;

});

}

);

}

supabase

.on(
'postgres_changes',
{
event:'*',
schema:'public',
table:'donors'
},
()=>{
loadDonors();
}
)
.subscribe();
